        // --- 1. CAS Mathematical Engine Logic ---
        let state = {
            frequency: 1.2,
            adaptation: 0.40,
            reinsurance: 0.85,
            fragility: 0.20,
            premium: 0.25,
            insurabilityIndex: 0.82
        };

        function calculateCASStep() {
            // Fragility increases with climate frequency, decreases with adaptation investment
            state.fragility = Math.min(1.0, Math.max(0.0, 
                state.fragility + (0.02 * state.frequency) - (0.03 * state.adaptation)
            ));

            // Premium rises with fragility and frequency, dampened by reinsurance
            state.premium = Math.min(2.0, Math.max(0.1,
                (0.25 * state.frequency) + (0.35 * state.fragility) + (0.20 * (1.0 - state.reinsurance))
            ));

            // Insurability Index (II_t) equation
            // II_t = w1*R - w2*P - w3*f - w4*I
            const ii = (0.45 * state.reinsurance) - (0.30 * state.premium) - (0.15 * (state.frequency / 4.0)) - (0.10 * state.fragility) + 0.65;
            state.insurabilityIndex = Math.min(1.0, Math.max(0.0, ii));

            updateUI();
        }

        // --- 2. WebGL / Three.js Filament Visualization ---
        let scene, camera, renderer, filamentMesh;

        const vertexShader = `
            uniform float uTime;
            uniform float uRisk;
            varying vec2 vUv;
            varying float vNoise;

            // Simple noise function
            float noise(vec3 p) {
                return sin(p.x * 3.0 + uTime) * cos(p.y * 3.0 + uTime) * sin(p.z * 3.0 + uTime);
            }

            void main() {
                vUv = uv;
                vec3 p = position;
                
                float n = noise(p);
                vNoise = n;
                
                // Displacement increases dramatically with risk index drop (1.0 - risk)
                float displacement = n * (1.0 - uRisk) * 0.8;
                p += normal * displacement;

                gl_Position = projectionMatrix * modelViewMatrix * vec4(p, 1.0);
            }
        `;

        const fragmentShader = `
            uniform float uTime;
            uniform float uRisk;
            varying vec2 vUv;
            varying float vNoise;

            void main() {
                // Color transition: Cyan (Stable) -> Amber -> Bright Red (Crisis)
                vec3 cyanColor = vec3(0.02, 0.71, 0.83);
                vec3 amberColor = vec3(0.95, 0.60, 0.10);
                vec3 redColor = vec3(0.93, 0.20, 0.20);

                vec3 finalColor;
                if (uRisk > 0.65) {
                    finalColor = mix(amberColor, cyanColor, (uRisk - 0.65) / 0.35);
                } else {
                    finalColor = mix(redColor, amberColor, uRisk / 0.65);
                }

                // Add glowing neural pulse
                float pulse = sin(vUv.x * 40.0 - uTime * 3.0) * 0.5 + 0.5;
                finalColor += pulse * 0.25;

                // Add flickering glitch when risk is critical
                if (uRisk < 0.40) {
                    float glitch = step(0.85, sin(uTime * 25.0));
                    finalColor += vec3(glitch * 0.4);
                }

                gl_FragColor = vec4(finalColor, 0.85);
            }
        `;

        function initWebGL() {
            const container = document.getElementById('canvas-container');
            
            scene = new THREE.Scene();
            camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000);
            camera.position.z = 4.5;

            renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
            renderer.setSize(window.innerWidth, window.innerHeight);
            renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
            container.appendChild(renderer.domElement);

            // Create Dynamic Torus Knot as Filament Representation
            const geometry = new THREE.TorusKnotGeometry(1.2, 0.35, 180, 30);
            const material = new THREE.ShaderMaterial({
                vertexShader: vertexShader,
                fragmentShader: fragmentShader,
                uniforms: {
                    uTime: { value: 0 },
                    uRisk: { value: state.insurabilityIndex }
                },
                wireframe: true,
                transparent: true
            });

            filamentMesh = new THREE.Mesh(geometry, material);
            scene.add(filamentMesh);

            // Ambient background particles
            const starGeo = new THREE.BufferGeometry();
            const starCount = 2000;
            const posArray = new Float32Array(starCount * 3);

            for(let i = 0; i < starCount * 3; i++) {
                posArray[i] = (Math.random() - 0.5) * 20;
            }

            starGeo.setAttribute('position', new THREE.BufferAttribute(posArray, 3));
            const starMat = new THREE.PointsMaterial({
                size: 0.015,
                color: 0x38bdf8,
                transparent: true,
                opacity: 0.4
            });

            const starMesh = new THREE.Points(starGeo, starMat);
            scene.add(starMesh);

            window.addEventListener('resize', onWindowResize);
            animate();
        }

        function animate(time) {
            requestAnimationFrame(animate);

            const t = time * 0.001;
            if (filamentMesh) {
                filamentMesh.material.uniforms.uTime.value = t;
                filamentMesh.material.uniforms.uRisk.value = state.insurabilityIndex;
                
                // Rotation speeds up under high risk
                const rotSpeed = 0.2 + (1.0 - state.insurabilityIndex) * 0.5;
                filamentMesh.rotation.x = t * rotSpeed * 0.5;
                filamentMesh.rotation.y = t * rotSpeed;
            }

            renderer.render(scene, camera);
        }

        function onWindowResize() {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        }
